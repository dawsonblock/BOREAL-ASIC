// Consolidated BCI Math Pipeline (Apex v4 + Kalman + Symbolic)
module apex_math_core #(
    parameter W = 16,
    parameter CH = 8
)(
    input  wire clk, rst, valid_in,
    input  wire safety_freeze,
    input  wire signed [W-1:0] feat [0:CH-1], // Simplified 1-band for TB
    input  wire signed [W-1:0] wL [0:CH-1],
    input  wire signed [W-1:0] wR [0:CH-1],
    input  wire signed [W-1:0] alpha, beta, ksig,
    
    output reg  signed [W-1:0] ux_out, uy_out, conf_out,
    output reg  [2:0]          token_out,
    output reg                 variance_spike
);
    // --- Apex v4 Core ---
    reg signed [W-1:0] B [0:CH-1];
    reg signed [W-1:0] N;
    reg signed [31:0] Lacc, Racc, varacc;
    reg signed [W-1:0] ux_raw, uy_raw;
    integer i;

    // --- Kalman State ---
    reg signed [W-1:0] s [0:4]; // ux, uy, vx, vy, drift
    reg signed [W-1:0] sp [0:4]; // Predicted
    reg signed [W-1:0] eux, euy;
    
    // Q3.13 precomputed constants
    localparam signed [15:0] K00 = 16'sd4096, K01 = 16'sd0;
    localparam signed [15:0] K10 = 16'sd0,    K11 = 16'sd4096;

    // --- Symbolic State ---
    reg [7:0] dwell;
    localparam SYM_TH = 16'sd1000;

    always @(posedge clk) begin
        if (rst) begin
            N <= 0; variance_spike <= 0; token_out <= 0; dwell <= 0;
            ux_out <= 0; uy_out <= 0; conf_out <= 0;
            for(i=0; i<CH; i=i+1) B[i] <= 0;
            for(i=0; i<5; i=i+1) s[i] <= 0;
        end else if (valid_in) begin
            // 1. APEX CORE
            Lacc = 0; Racc = 0; varacc = 0;
            for(i=0; i<CH; i=i+1) begin
                B[i] <= B[i] + ((alpha * (feat[i] - B[i])) >>> (W-2));
                Lacc = Lacc + (wL[i] * (feat[i] - B[i]));
                Racc = Racc + (wR[i] * (feat[i] - B[i]));
                varacc = varacc + (((feat[i] - B[i]) * (feat[i] - B[i])) >>> (W-2));
            end
            N <= N + ((beta * (varacc[W-1:0] - N)) >>> (W-2));
            variance_spike <= (varacc > 32'sd100000); // Safety flag

            ux_raw = (Lacc[W-1:0] - Racc[W-1:0]);
            uy_raw = -(varacc[W-1:0]);
            
            // Confidence
            if ((ux_raw[W-1]? -ux_raw : ux_raw) > N)
                conf_out <= ((ux_raw[W-1]? -ux_raw : ux_raw) - N) / ksig;
            else conf_out <= 0;

            // 2. KALMAN (Separated predict/update for non-blocking safety)
            sp[0] = s[0] + s[2]; sp[1] = s[1] + s[3];
            sp[2] = s[2];        sp[3] = s[3]; 
            sp[4] = s[4];

            eux = ux_raw - sp[0]; euy = uy_raw - sp[1];

            s[0] <= sp[0] + ((K00 * eux + K01 * euy) >>> 13);
            s[1] <= sp[1] + ((K10 * eux + K11 * euy) >>> 13);
            s[2] <= sp[2]; s[3] <= sp[3]; s[4] <= sp[4];

            if (safety_freeze) begin
                ux_out <= 0; uy_out <= 0;
            end else begin
                ux_out <= s[0]; uy_out <= s[1];
            end

            // 3. SYMBOLIC DECODER
            if (!safety_freeze) begin
                if (conf_out > SYM_TH && ux_out > SYM_TH) begin
                    dwell <= dwell + 1;
                    if (dwell > 50) token_out <= 3; // CONFIRM
                end else if (conf_out > SYM_TH && ux_out < -SYM_TH) begin
                    token_out <= 1; // CLICK
                    dwell <= 0;
                end else begin
                    token_out <= 0; dwell <= 0;
                end
            end else begin
                token_out <= 0;
            end
        end
    end
endmodule
