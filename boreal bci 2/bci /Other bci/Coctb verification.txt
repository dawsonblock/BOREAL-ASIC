import cocotb
from cocotb.clock import Clock
from cocotb.triggers import RisingEdge, Timer
import random

async def simulate_spi_frame(dut, data_words, missing=False):
    """Simulates the ADS1299 SPI frame generation"""
    if missing:
        await Timer(1.5, units='ms') # Exceeds 1ms timeout
        return

    dut.drdy_n.value = 0
    await RisingEdge(dut.clk)
    dut.drdy_n.value = 1
    
    # Wait for CS_n to drop
    while dut.cs_n.value == 1:
        await RisingEdge(dut.clk)
    
    # 216 bits: 24-bit status + 8 * 24-bit channels
    frame_bits = "000000000000000000000000" # Status
    for word in data_words:
        frame_bits += format(word & 0xFFFFFF, '024b')
    
    for bit in frame_bits:
        dut.miso.value = int(bit)
        # Wait for 1 complete SCLK cycle (driven by DUT)
        await RisingEdge(dut.sclk)
        await Timer(1, units='ns') # hold

@cocotb.test()
async def test_safety_escalations(dut):
    """Test standard EEG parsing and fault/safety tier escalation"""
    cocotb.start_soon(Clock(dut.clk, 20, units="ns").start()) # 50 MHz
    
    dut.rst.value = 1
    dut.clear_latch.value = 0
    dut.drdy_n.value = 1
    dut.miso.value = 0
    await Timer(100, units="ns")
    dut.rst.value = 0
    await Timer(100, units="ns")

    dut._log.info("--- Test 1: Normal Operation (Tier 0) ---")
    # Feed normal low-variance sinusoids
    for _ in range(5):
        normal_data = [random.randint(1000, 5000) for _ in range(8)]
        await simulate_spi_frame(dut, normal_data)
        await Timer(5, units="us")
        
    assert dut.safety_tier.value == 0, f"Expected Tier 0, got {dut.safety_tier.value}"

    dut._log.info("--- Test 2: Artifact Injection -> Tier 2 (Freeze) ---")
    # Feed massive spike to trigger variance_spike flag
    spike_data = [8000000 for _ in range(8)]
    await simulate_spi_frame(dut, spike_data)
    await Timer(5, units="us")
    
    assert dut.safety_tier.value == 2, f"Expected Tier 2 (Freeze), got {dut.safety_tier.value}"
    assert dut.hid_ux.value == 0, "Motion not frozen during Tier 2!"

    dut._log.info("--- Test 3: Latch Clear Recovery ---")
    dut.clear_latch.value = 1
    await RisingEdge(dut.clk)
    dut.clear_latch.value = 0
    await Timer(1, units="us")
    assert dut.safety_tier.value == 0, "Failed to clear latch"

    dut._log.info("--- Test 4: Missing Frame Timeout -> Tier 3 (Hard Stop) ---")
    await simulate_spi_frame(dut, [], missing=True)
    assert dut.safety_tier.value == 3, f"Expected Tier 3 (Hard Stop), got {dut.safety_tier.value}"

    dut._log.info("All hardware safety constraints verified successfully.")
